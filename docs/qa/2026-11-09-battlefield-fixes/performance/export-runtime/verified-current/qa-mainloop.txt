func _initialize() -> void:

		return
	create_timer(850).timeout.connect(func(): quit(2))
	var args := OS.get_cmdline_user_args()
	if args.size()>3 and args[3]=="diagnostic":
		qa_diagnostic(args[0])
		return
	output = args[0]
	phone = args.size() > 2 and args[2] == "phone"
	touch_profile = phone or (args.size() > 2 and args[2] == "tablet")
	mode = args[3] if args.size() > 3 else "load"
	sample_n = clampi(int(args[4]), 30, 600) if args.size() > 4 else 180
	profile_cpu = OS.get_environment("BINDU_PROFILE_CPU") == "1"
	Tutorial.loaded = true
	Tutorial.done = true
	Tutorial.play_done = true
	Meta.games = 10
	Meta.guide_on = false
	Meta.sim_phone = touch_profile
	Meta.sim_safe = Rect2(86, 0, 86, 38) if phone else Rect2()
	Meta.skill_auto = false
	Meta.skill_aim_slow = false
	DisplayServer.window_set_vsync_mode(DisplayServer.VSYNC_DISABLED)
	Engine.max_fps = 0
	main = load("res://main.tscn").instantiate()
	VirtualInput.mount(main, root)
	await process_frame
	main.progression_system._hide_title()
	main.set_process(false)
	main.auto_start = false
	Meta.ui_scale = float(args[1])
	main.touch_ui = touch_profile
	main.hud_system._apply_ui_scale()
	render_rid = main.get_viewport().get_viewport_rid()

		await layout_cases()

		for speed in [1.0, 8.0]:
			for reduced in [false]:
				await measure_case(speed, reduced)

	var data := {"queue_probes":qa_queue_rows,"runtime_features":{"template":OS.has_feature("template"),"debug_build":OS.is_debug_build(),"editor":OS.has_feature("editor"),"tracking_setting":ProjectSettings.get_setting("debug/settings/gdscript/always_track_call_stacks")},"base_revision":OS.get_environment("BINDU_AUDIT_REVISION"), "source_state":OS.get_environment("BINDU_LOAD_SOURCE_STATE"), "engine":Engine.get_version_info(), "renderer":RenderingServer.get_video_adapter_name(), "display_server":DisplayServer.get_name(), "physical_size":str(root.size), "logical_size":str(main.get_viewport_rect().size), "ui_scale":Meta.ui_scale, "phone_simulation":phone, "touch_profile":touch_profile, "mode":mode, "fixed_simulation_delta":1.0 / 60.0, "sample_frames_per_condition":sample_n, "warmup_frames_per_condition":30, "profiler_enabled":profile_cpu, "render_drive":"RenderingServer.force_draw() per sample; return timing, not physical presentation", "source_hashes":source_hashes, "measurements":measurements, "assertion_failures":fails, "ui_checker_messages":ui_errors}

func _hash_sources(path: String) -> void:
	for filename in DirAccess.get_files_at(path):
		if filename.ends_with(".gd"):
			var p := path.path_join(filename)
			source_hashes[p] = FileAccess.get_sha256(p)
	for directory in DirAccess.get_directories_at(path):
		_hash_sources(path.path_join(directory))

func fill_orbs() -> void:
	var alive: int = main.orbs.get_children().filter(func(o): return is_instance_valid(o) and not o.is_queued_for_deletion() and o.hp > 0.0).size()
	for i in maxi(0, orb_goal - alive):
		var index := orb_serial % orb_goal
		orb_serial += 1
		var p := Vector2(90.0 + (index % 25) * 44.0, 105.0 + floori(index / 25.0) * 43.0)
		if p.distance_to(main.viewport_system.core()) < 70.0:
			p.y += 100.0
		_orb(p, 10000000.0, Orb.Kind.FAST if index % 5 == 0 else Orb.Kind.NORMAL, 8.0)

func measure_case(speed: float, reduced: bool) -> void:
	await load_fixture()
	main.game_speed = speed
	Meta.fx_low = reduced
	for i in 30:
		await process_frame
		main.frame_system._process(1.0 / 60.0)
		fill_orbs()
		RenderingServer.force_draw()
	main.combat_system.prof.clear()
	var wall: Array = []
	var intervals: Array = []
	var simulation: Array = []
	var render_cpu: Array = []
	var render_gpu: Array = []
	var orbs: Array = []
	var projectiles: Array = []
	var zones: Array = []
	var probes: Array = []
	var condition_start := Time.get_ticks_usec()
	var previous_draw := condition_start
	for i in sample_n:
		await process_frame
		var start := Time.get_ticks_usec()
		main.frame_system._process(1.0 / 60.0)
		var after_sim := Time.get_ticks_usec()
		fill_orbs()
		RenderingServer.force_draw()
		var after_draw := Time.get_ticks_usec()
		wall.append((after_draw - start) / 1000.0)
		intervals.append((after_draw - previous_draw) / 1000.0)
		previous_draw = after_draw
		simulation.append((after_sim - start) / 1000.0)
		render_cpu.append(RenderingServer.viewport_get_measured_render_time_cpu(render_rid))
		render_gpu.append(RenderingServer.viewport_get_measured_render_time_gpu(render_rid))
		orbs.append(main.live_orbs.size())
		projectiles.append(main.projectiles.get_child_count())
		zones.append(main.zones.get_child_count())
		if i % 60 == 30:
			var before_probe: Dictionary = main.combat_system.prof.duplicate()
			probes.append(await pause_probe())
			main.combat_system.prof = before_probe
			previous_draw = Time.get_ticks_usec()  # 명시적 정지/재개 프로브의 추가 프레임은 일반 간격에서 제외
	var label := "x%d-fx-%s" % [roundi(speed), "reduced" if reduced else "full"]
	var parts := {}
	for key in main.combat_system.prof:
		parts[key] = float(main.combat_system.prof[key]) / 1000.0 / sample_n
	var row := {"condition":label, "speed":speed, "fx_low":reduced, "samples":wall.size(), "elapsed_ms":(Time.get_ticks_usec() - condition_start) / 1000.0, "simulation_game_seconds":sample_n * speed / 60.0, "forced_render_interval_ms":stats(intervals), "simulation_to_force_draw_return_ms":stats(wall), "main_frame_cpu_ms":stats(simulation), "viewport_render_cpu_ms":stats(render_cpu), "viewport_render_gpu_ms":stats(render_gpu), "profiled_parts_ms_per_sample":parts, "mean_orbs":mean(orbs), "mean_projectiles":mean(projectiles), "mean_zones":mean(zones), "pause_probes":probes, "raw_forced_render_interval_ms":intervals, "raw_main_frame_cpu_ms":simulation, "raw_viewport_render_cpu_ms":render_cpu}
	measurements.append(row)
	print("MEASURE ", JSON.stringify(row))
	await snap(label)

func pause_probe() -> Dictionary:
	var e := InputEventKey.new()
	e.keycode = KEY_P
	e.pressed = true
	var start := Time.get_ticks_usec()
	VirtualInput.send(e)
	var dispatch_ms := (Time.get_ticks_usec() - start) / 1000.0
	var state_seen: bool = paused
	e.pressed = false
	VirtualInput.send(e)
	main.frame_system._process(0.0)
	RenderingServer.force_draw()
	var drawn_ms := (Time.get_ticks_usec() - start) / 1000.0
	var pause_visible: bool = main.pause_box.is_visible_in_tree()
	ok("BF-49", state_seen and pause_visible, "부하 중 가상 P 입력의 정지 상태·메뉴 표시")
	e.pressed = true
	VirtualInput.send(e)
	e.pressed = false
	VirtualInput.send(e)
	main.frame_system._process(0.0)
	RenderingServer.force_draw()
	return {"dispatch_to_state_ms":dispatch_ms, "dispatch_to_force_draw_return_ms":drawn_ms, "paused_state_seen":state_seen, "pause_ui_seen":pause_visible}

func mean(values: Array) -> float:
	var total := 0.0
	for value in values:
		total += float(value)
	return total / maxf(1.0, values.size())

func stats(values: Array) -> Dictionary:
	var sorted := values.duplicate()
	sorted.sort()
	return {"mean":mean(values), "median":sorted[sorted.size() / 2], "p95":sorted[mini(sorted.size() - 1, floori(sorted.size() * 0.95))], "max":sorted[-1], "nonzero_samples":values.filter(func(v): return v > 0.0).size()}

var qa_queue_rows := []
var qa_delivery := {}

func qa_queue_probes() -> void:
	for speed in [1.0,8.0]:
		await load_fixture()
		main.game_speed = speed
		Meta.fx_low = false
		for i in 30:
			await process_frame
			main.frame_system._process(1.0/60.0)
			fill_orbs()
			RenderingServer.force_draw()
		for trial in 3:
			await process_frame
			qa_delivery = {}
			var worker := Thread.new()
			var start := Time.get_ticks_usec()
			worker.start(qa_post.bind(trial))
			main.frame_system._process(1.0/60.0)
			var compute_end := Time.get_ticks_usec()
			fill_orbs()
			RenderingServer.force_draw()
			var draw_end := Time.get_ticks_usec()
			var deadline := Time.get_ticks_usec()+2000000
			while qa_delivery.is_empty() and Time.get_ticks_usec()<deadline:
				await create_timer(0.001).timeout
			worker.wait_to_finish()
			ok("BF-49",not qa_delivery.is_empty(),"queued synthetic pause request delivered")
			if qa_delivery.is_empty(): continue
			var row: Dictionary = qa_delivery.duplicate()
			row.merge({"speed":speed,"trial":trial,"main_frame_cpu_ms":(compute_end-start)/1000.0,"draw_return_ms":(draw_end-start)/1000.0,"posted_during_compute":row.posted_us>=start and row.posted_us<compute_end,"posted_before_draw_return":row.posted_us<draw_end})
			qa_queue_rows.append(row)
			ok("BF-49",row.paused_state_seen and row.pause_ui_seen,"queued pause reached state and menu")
			var resume := InputEventKey.new()
			resume.keycode = KEY_P
			resume.pressed = true
			VirtualInput.send(resume)
			resume.pressed = false
			VirtualInput.send(resume)
			ok("BF-49",not paused,"queued probe resumed")
			await process_frame

func qa_post(id: int) -> void:
	OS.delay_msec(5)
	call_deferred("qa_deliver",id,Time.get_ticks_usec())

func qa_deliver(id: int, posted_us: int) -> void:
	var received := Time.get_ticks_usec()
	var press := InputEventKey.new()
	press.keycode = KEY_P
	press.pressed = true
	VirtualInput.send(press)
	var done := Time.get_ticks_usec()
	qa_delivery = {"id":id,"posted_us":posted_us,"received_us":received,"queue_to_dispatch_ms":(received-posted_us)/1000.0,"dispatch_to_state_ms":(done-received)/1000.0,"paused_state_seen":paused,"pause_ui_seen":main.pause_box.is_visible_in_tree()}
	press.pressed = false
	VirtualInput.send(press)

func qa_diagnostic(output_dir: String) -> void:
	DirAccess.make_dir_recursive_absolute(output_dir)
	ErrLog.install()
	qa_error_root()
	var rows := ErrLog.take()
	var selected := rows.filter(func(row):return "BinduQA controlled trace probe" in str(row.get("msg","")))
	var bt := str(selected[0].get("bt","")) if selected.size()==1 else ""
	var result := {"events":selected,"expected_events":selected.size()==1,"root_in_backtrace":"qa_error_root" in bt,"mid_in_backtrace":"qa_error_mid" in bt,"leaf_in_backtrace":"qa_error_leaf" in bt,"debug_build":OS.is_debug_build(),"template":OS.has_feature("template"),"tracking_setting":ProjectSettings.get_setting("debug/settings/gdscript/always_track_call_stacks")}
	FileAccess.open(output_dir.path_join("diagnostic.json"),FileAccess.WRITE).store_string(JSON.stringify(result,"\t"))
	print("QA DIAGNOSTIC ",JSON.stringify(result))
	quit(0 if selected.size()==1 else 1)

func qa_error_root() -> void:
	qa_error_mid()

func qa_error_mid() -> void:
	qa_error_leaf()

func qa_error_leaf() -> void:
	push_error("BinduQA controlled trace probe")
